```python
import statistics, math

# 26/06/29〜26/07/28の日次終値（ascending、kabutan.jp確定終値ページより、curl取得 2026-07-29 06:43:45Z）
dates  = ['06/29','06/30','07/01','07/02','07/03','07/06','07/07','07/08','07/09','07/10',
          '07/13','07/14','07/15','07/16','07/17','07/21','07/22','07/23','07/24','07/27','07/28']
closes = [606,   601,   604,   617,   621,   628,   644,   631,   629,   630,
          623,   624,   630,   624,   617,   628,   632,   643,   642,   652,   643]
# 07/28以前20営業日（06/30〜07/28）の出来高
vols20 = [206800,166300,307000,201800,216600,318200,137000,98400,194400,134000,
          120100,115200,200600,218000,294900,156400,278100,175800,197900,117600]

avgvol20 = statistics.mean(vols20)
print("20日平均出来高:", avgvol20, "→1.5倍:", avgvol20*1.5)

rets = [(closes[i]-closes[i-1])/closes[i-1] for i in range(1, len(closes))]
sigma20 = statistics.stdev(rets) * 100
print("σ20(%):", round(sigma20, 4))

base_high = 643  # 5408-T2原文の基準高値（2026-07-23確定終値）
pct2 = base_high * 0.02
sigma_abs = base_high * sigma20 / 100
threshold = math.ceil(base_high + max(pct2, sigma_abs, 1))
print("2%成分:", round(pct2,2), "σ20成分:", round(sigma_abs,2), "実務閾値(ceil):", threshold)

close_today, vol_today = 659, 408600
print("価格条件:", close_today, ">=", threshold, "→", close_today >= threshold)
print("出来高倍率:", round(vol_today/avgvol20, 4), "→ 1.5倍以上:", vol_today >= avgvol20*1.5)
```
