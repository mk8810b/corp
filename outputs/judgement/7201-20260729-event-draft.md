# kabutan.jp確定終値ページ（curl取得・UA指定、取得日時2026-07-29 06:44 UTC）の日次データ
# (date, close, volume)。直近20営業日窓＝2026-06-30〜2026-07-28（07-29は含めない）。
rows_desc = [  # 日付降順（ページ表示順）
    ('2026-07-28', 337.4, 36177500), ('2026-07-27', 332.9, 28384500),
    ('2026-07-24', 322.2, 31096500), ('2026-07-23', 332.3, 19174800),
    ('2026-07-22', 334.3, 29593600), ('2026-07-21', 332.3, 26456800),
    ('2026-07-17', 329.3, 42164800), ('2026-07-16', 329.0, 43093300),

    ('2026-07-07', 321.8, 34389800), ('2026-07-06', 319.1, 35392500),
    ('2026-07-03', 313.1, 28248500), ('2026-07-02', 312.0, 35074300),
