## 13. Python再計算（憲法第2章-5: 数値の再計算は必ずPythonで行う）

以下は本メモで用いた全数値の算出コードである。**入力の日足30行は0-1節でcurl取得したHTMLから
機械パースしたものであり、記憶ベース・推測ベースの数値は一切含まない**（第2条・INC-001の教訓）。

# --- 入力1: kabutan.jp 日足（9889）。取得元 https://kabutan.jp/stock/kabuka?code=9889&ashi=day&page=1
# 取得日時 2026-08-03T07:30:08Z / 07:31:22Z / 07:33:04Z / 07:34:48Z（4回・全項目一致）
# (日付, 終値, 出来高) 昇順。σ20/出来高平均の窓は判定日08-03を「除く」07/03〜07/31の20営業日。
win = [  # 2026-07-03 〜 2026-07-31（20営業日。07-20は海の日で非営業日）
 ("07-03",1332,202300),("07-06",1365,268600),("07-07",1376,200900),("07-08",1396,225900),
 ("07-09",1405,192500),("07-10",1409,124600),("07-13",1404,260300),("07-14",1447,234500),
 ("07-15",1434,111200),("07-16",1443, 96900),("07-17",1427,131100),("07-21",1454,127600),
 ("07-22",1443,141300),("07-23",1428,166000),("07-24",1450,211600),("07-27",1545,214200),
 ("07-28",1574,274600),("07-29",1679,420700),("07-30",1651,286100),("07-31",1616,479200)]
prev_close = 1285          # 窓外の前営業日 2026-07-02 の終値

assert abs((base_close/prev_day_close-1)*100 - 4.9505) < 1e-3   # +4.9505% ≒ kabutan表示 +4.95%

# --- 2. 出来高条件（T1）
vols = [v for _,_,v in win]
avg20 = sum(vols)/len(vols)                    # => 218,505.0 株（合計 4,370,100 株）

ratio = base_vol/avg20                         # => 2.2759 倍
assert base_vol >= thr15                       # 出来高条件 成立

# --- 3. 価格条件（T1）
assert base_close >= 1655                      # 価格条件 成立
margin = (base_close/1655-1)*100               # => +2.4773%

# --- 4. σ20（新基準日08-03用。窓は判定日を除く07/03〜07/31）
closes = [prev_close] + [c for _,c,_ in win]
rets = [(closes[i]/closes[i-1]-1)*100 for i in range(1,len(closes))]
sigma_pop = st.pstdev(rets)                    # => 2.373731 %
sigma_smp = st.stdev(rets)                     # => 2.435397 %  ← 拘束的ケースにつき正本
# --- 5. D-018ゲートと新トリガー閾値
two_pct = base_close*0.02                      # => 33.9200 円
sig_amt = base_close*sigma_smp/100             # => 41.3043 円  ← 拘束成分
gate = max(two_pct, sig_amt, 1.0)              # => 41.3043 円
t9_up   = math.ceil (base_close+gate)          # => 1,738 円（raw 1,737.3043）
t8_down = math.floor(base_close-gate)          # => 1,654 円（raw 1,654.6957）
# 母系列での感応度（参考）: gate=40.2585 → 上1,737円 / 下1,655円（各1円差）

# --- 6. 押し目トリガー（T3）の基準高値と帯
base_high = max([c for _,c,_ in win] + [base_close])   # => 1,696 円（本日終値が最高値）
band_lo = math.ceil (base_high*0.85)           # => 1,442 円（実効 -14.9764%、帯の内側）
band_hi = math.floor(base_high*0.95)           # => 1,611 円（実効  -5.0118%、帯の内側）

# --- 7. バリュエーション（BPS/EPS/DPSは27.3期1Q決算短信原本由来・朝の確定版メモ12節で再検算済み）
BPS, EPS, DPS = 405.6937, 99.57, 60.0
for px in (1616, 1696):
    per, pbr, yld = px/EPS, px/BPS, DPS/px*100
# 1,616円 → PER 16.2298 / PBR 3.9833 / 利回り 3.7129%
# 1,696円 → PER 17.0332 / PBR 4.1805 / 利回り 3.5377%   （kabutan掲載 17.1/4.18/3.54 と一致）
t7_pbr3 = math.floor(BPS*3.00)                 # => 1,217 円（9889-T7、BPS不変につき閾値も不変）
dist_t7 = (1696/1217-1)*100                    # => +39.3591 %

# --- 8. サイジング（D-011/D-019/D-015、分母は現在資金残高＝D-015比例ベット方式）
cash = 3_000_000            # corp/cash.md（2026-07-09申告。以後CEOの実申告なし）
held_mv = 0                 # corp/portfolio.md 状態=HELD 0件

unit_cost = 1696*100                            # => 169,600 円（kabutan「売買最低代金」と一致）
ratio_funds = unit_cost/funds*100               # => 5.6533 %
prob2, prob3 = funds*0.02, funds*0.03           # => 60,000 円 / 90,000 円
excess3 = unit_cost/prob3*100-100               # => +88.4444 %  ⇒ 実弾見送り（シャドー追跡のみ）

# --- 9. 先行上昇・相対リターン
run6w = (1696/1164-1)*100                       # => +45.7045 %（2026-06-19 終値1,164円起点）
d2     = (1696/1651-1)*100                      # => +2.7256 %（開示前日07-30起点の2営業日累計）
nk = {"07-30":61867.43, "07-31":64362.02, "08-03":63754.90}   # 日経平均（kabutan code=0000）
nk_1d = (nk["08-03"]/nk["07-31"]-1)*100         # => -0.9433 %
nk_2d = (nk["08-03"]/nk["07-30"]-1)*100         # => +3.0508 %
peers = {  # (07-30終値, 07-31終値, 08-03終値)
 "3626 TIS":(3937,3780,3712), "2327 NSSOL":(3931,3915,3726), "4812 電通総研":(2714,2687,2700),
 "9682 DTS":(1174,1158,1080), "3916 DIT":(965,947,923), "4662 フォーカス":(1774,1748,1731)}
p1 = st.median([(c3/c31-1)*100 for c30,c31,c3 in peers.values()])   # => -2.1666 %

rel = ((4.9505-nk_1d), (d2-nk_2d), (4.9505-p1), (d2-p2))
# => 対日経 単日 +5.8938pt / 2営業日 -0.3252pt ; 対同業中央値 単日 +7.1171pt / 2営業日 +7.5093pt

# --- 10. 売買の質（弱い傍証）
lot_0731, lot_0803 = 479200/1972, 497300/1633   # => 243.00 株/回 → 304.53 株/回（+25.32%）
vwap_gap = (1696/1687.967-1)*100                # => +0.4759 %（終値がVWAPを上回る）

# --- 11. 本日の開示（0-3節）と自己株取得の試算（T6）
issued, treasury, rs = 71_094_972, 10_300_331, 242_700
ex_old = issued - treasury                      # => 60,794,641 株

    n = min(cap_yen//px, cap_sh)
# 1,616円 → 618,811株（ex_new比 1.0138%） / 1,696円 → 589,622株（ex_new比 0.9660%、-4.7169%）

# --- 12. D-002バリア参考値（呼び値1円でceiling）

# 1,654 → 1,489 / 2,068 ； 1,696 → 1,527 / 2,120 ； 1,738 → 1,565 / 2,173
