gate_incl = max(two_pct, sigma_incl_yen, 1)

thr_excl = math.ceil(thr_excl_raw)
thr_incl = math.ceil(thr_incl_raw)
vol_thr = math.ceil(avg_vol20_excl*1.5)

print(f"2%成分={two_pct:.4f}円 σ20(除く)成分={sigma_excl_yen:.4f}円 σ20(含む)成分={sigma_incl_yen:.4f}円")
