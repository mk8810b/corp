def sig(cl):
    simp = [cl[i]/cl[i-1]-1 for i in range(1, len(cl))]
    logr = [math.log(cl[i]/cl[i-1]) for i in range(1, len(cl))]
    return (st.pstdev(simp)*100, st.stdev(simp)*100, st.pstdev(logr)*100, st.stdev(logr)*100)
